// Hint: Type return [i, map.get(comp)]

function twoSum(nums, target) {
  const map = new Map();
  for (let i = 0; i < nums.length; i++) {
    const comp = target - nums[i];
    if (map.has(comp)) {
      return [map.get(comp), i];
    }
    map.set(nums[i], i);
  }
}

    map = {}
    for i, n in enumerate(nums):
        if target - n in map:
            return [map[target-n], i]
        map[n] = i
